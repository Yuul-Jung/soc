SECTION 02

Polling에서 Interrupt로

Interrupt가 무조건 더 빠르다는 뜻은 아니다. 핵심은 CPU가 완료 여부만 확인하기 위해 지속적으로 STATUS Register를 읽을 필요가 없다는 점이다.

SECTION 03

전체 Architecture

NPU DONE/ERROR → IRQ_STATUS → IRQ_ENABLE → irq → AXI Interrupt Controller → MicroBlaze V → ISR
SECTION 04

왜 DONE Pulse를 직접 연결하지 않는가

Accelerator의 DONE은 짧은 Event일 수 있다. CPU/Interrupt Controller가 처리하기 전에 Pulse가 끝나면 Software가 Event를 복구하기 어렵다. 따라서 Peripheral 내부 Status Register에 Event를 Latched한다.

SECTION 05

Level IRQ 구조

irq_o → OR(IRQ_STATUS AND IRQ_ENABLE)

Pending Source가 Enable된 동안 IRQ는 High 상태를 유지한다. ISR이 Pending Bit를 Clear한 뒤에야 IRQ가 Low로 내려간다.

SECTION 07

IRQ_ENABLE

Event를 저장할지 여부가 아니라 CPU에 IRQ를 전달할지를 제어한다. Enable이 0이어도 Pending은 저장될 수 있다.

SECTION 08

IRQ_STATUS W1C

W1C(Write-One-to-Clear, 1을 써서 클리어)는 해당 Bit에 1을 Write하면 Pending을 Clear하는 방식이다.

DONE pending clear  → write 0x00000001
ERROR pending clear → write 0x00000002
둘 다 clear         → write 0x00000003
SECTION 09

Pending과 Enable의 차이

IRQ_STATUS → 사건이 발생했는가? IRQ_ENABLE → 그 사건을 CPU Interrupt로 전달할 것인가?

DONE Event가 발생한 뒤 Enable을 켜도 Pending이 남아 있으면 IRQ가 바로 High가 된다.

SECTION 10

Event-set 우선

irq_status_next
→ (old_status AND NOT clear_mask) OR event_set

Clear와 새 Event가 같은 Clock에 겹쳐도 Event가 사라지지 않도록 Set을 마지막 우선순위로 둔다.

SECTION 11

DONE Interrupt

accel_done → IRQ_STATUS.DONE=1 → IRQ_ENABLE.DONE=1 → irq=1
SECTION 12

ERROR Interrupt

잘못된 Command → error_event → IRQ_STATUS.ERROR=1 → irq

이번 장의 Error Event는 SoC 31과 동일하게 BUSY 중 재START를 대표 사례로 사용한다.

SECTION 13

START while BUSY

BUSY 중 START는 수락하지 않으며 STATUS.ERROR와 IRQ_STATUS.ERROR를 각각 Latched한다. 일반 Status Sticky와 Interrupt Pending은 독립적으로 Clear할 수 있다.

SECTION 14

AXI Write Commit 보강

AW와 W를 독립 Capture한 뒤 둘 다 Pending일 때 Register Write와 BVALID를 같은 Commit Clock에 처리한다. Write Response가 Register Side Effect보다 앞서지 않도록 한 구조다.

// ------------------------------------------------------------
// SoC 32 · NPU Interrupt Register Bank
// ------------------------------------------------------------
//
// Existing SoC 31 map:
//
// 0x00 CONTROL      W1P
// 0x04 STATUS       RO
// 0x08 VERSION      RO
// 0x0C SAT_COUNT    RO
// 0x10 BIAS_CH0     RW
// 0x14 BIAS_CH1     RW
// 0x18 SCALE_CH0    RW
// 0x1C SCALE_CH1    RW
// 0x20 ZERO_POINT   RW
// 0x40..0x5C ACC_IN0..7 RW
// 0x60 OUT_CH0      RO
// 0x64 OUT_CH1      RO
// 0x68 CYCLE_COUNT  RO
//
// SoC 32 additions:
//
// 0x6C IRQ_ENABLE   RW
//      bit 0 DONE IRQ enable
//      bit 1 ERROR IRQ enable
//
// 0x70 IRQ_STATUS   RO / W1C
//      bit 0 DONE event pending
//      bit 1 ERROR event pending
//
// Level interrupt:
//
//   irq_o = |(IRQ_ENABLE & IRQ_STATUS)
//
// IRQ_STATUS remains set until software writes 1 to the
// corresponding bit. Event set wins over W1C clear if both
// happen in the same clock.

localparam [31:0] VERSION_VALUE = 32'h0001_0000;

function [31:0] apply_wstrb32;
    input [31:0] old_value;
    input [31:0] new_value;
    input [3:0]  strb;

    integer byte_index;

    begin
        apply_wstrb32 = old_value;

        for (
            byte_index = 0;
            byte_index < 4;
            byte_index = byte_index + 1
        ) begin
            if (strb[byte_index]) begin
                apply_wstrb32[
                    (byte_index * 8) +: 8
                ] =
                    new_value[
                        (byte_index * 8) +: 8
                    ];
            end
        end
    end
endfunction

    !w_pending_q &&
    !s_axi_bvalid;

assign s_axi_bresp = 2'b00;

wire w_handshake_w;

assign w_handshake_w =
    s_axi_wvalid &&
    s_axi_wready;

    .wr_data_i(wdata_q),
    .wr_strb_i(wstrb_q),

    .accel_busy_o(accel_busy_o),
    .accel_done_sticky_o(accel_done_sticky_o),

static volatile u32 NpuErrorFlag = 0U;

static void NpuIsr(void *CallbackRef)
{
    const UINTPTR base = (UINTPTR)CallbackRef;
    const u32 irq_status = Npu_GetIrqStatus(base);

    NpuLastIrqStatus = irq_status;

    if ((irq_status & NPU_IRQ_DONE_MASK) != 0U) {
        NpuDoneFlag = 1U;
    }

    if ((irq_status & NPU_IRQ_ERROR_MASK) != 0U) {
        NpuErrorFlag = 1U;
    }

    // Clear the peripheral source before returning.
    // IRQ_STATUS uses Write-One-to-Clear.
    Npu_ClearIrq(
        base,
        irq_status & NPU_IRQ_ALL_MASK
    );
}

static int SetupInterruptSystem(UINTPTR base)
{
    int status;

    // (변경) SDT는 BaseAddress, 비 SDT는 DeviceId를 사용한다.
#ifndef SDT
    status = XIntc_Initialize(&Intc, INTC_DEVICE_ID);
#else
    status = XIntc_Initialize(&Intc, INTC_BASEADDR);
#endif

    if (status != XST_SUCCESS) {
        return XST_FAILURE;
    }

    status = XIntc_Connect(
        &Intc,
        NPU_INTR_ID,
        (XInterruptHandler)NpuIsr,
        (void *)base
    );

    if (status != XST_SUCCESS) {
        return XST_FAILURE;
    }

    status = XIntc_Start(
        &Intc,
        XIN_REAL_MODE
    );

    if (status != XST_SUCCESS) {
        return XST_FAILURE;

        XIL_EXCEPTION_ID_INT,

        return 1;

// order:
//   0 = AW and W together
//   1 = AW first, W later
//   2 = W first, AW later
task automatic axi_write(
    input [7:0] address,
    input [31:0] data,
    input [3:0] strobe,
    input integer order
);
    bit aw_done;
    bit w_done;
    integer timeout;

    begin
        aw_done = 0;
        w_done = 0;

        @(negedge clk);

        awaddr = address;
        wdata = data;
        wstrb = strobe;

        case (order)
            0: begin
                awvalid = 1'b1;
                wvalid = 1'b1;
            end

            1: begin
                awvalid = 1'b1;
                wvalid = 1'b0;
            end

            default: begin
                awvalid = 1'b0;
                wvalid = 1'b1;
            end
        endcase

        timeout = 0;

        while (!(aw_done && w_done)) begin
            @(posedge clk);
            // (변경) READY가 갱신되기 전 상승 에지의 handshake를 관측한다.

            if (
                awvalid &&
                awready
            ) begin
                aw_done = 1;
            end

            if (
                wvalid &&
                wready
            ) begin
                w_done = 1;
            end

            @(negedge clk);

            if (aw_done) begin
                awvalid = 1'b0;
            end

            if (w_done) begin
                wvalid = 1'b0;
            end

            if (
                (order == 1) &&
                aw_done &&
                !w_done
            ) begin
                wvalid = 1'b1;
            end

            if (
                (order == 2) &&
                w_done &&
                !aw_done
            ) begin
                awvalid = 1'b1;
            end

            timeout = timeout + 1;

            if (timeout > 30) begin
                $fatal(1, "AXI write handshake timeout");
            end
        end

        timeout = 0;

        while (!bvalid) begin
            @(posedge clk);
            #1;

            timeout = timeout + 1;

            if (timeout > 30) begin
                $fatal(1, "AXI BVALID timeout");
            end
        end

        check_true(
            "AXI BRESP OKAY",
            bresp == 2'b00
        );

        // One extra clock ensures the B-channel handshake is complete.
        @(posedge clk);
        #1;
    end
endtask

task automatic axi_read(
    input [7:0] address,
    output [31:0] data
);
    integer timeout;

    begin
        @(negedge clk);

        araddr = address;
        arvalid = 1'b1;

        timeout = 0;

        while (!arready) begin
            @(posedge clk);
            #1;

            timeout = timeout + 1;

            if (timeout > 30) begin
                $fatal(1, "AXI ARREADY timeout");
            end
        end

        @(posedge clk);
        #1;

        if (!rvalid) begin
            timeout = 0;

            while (!rvalid) begin
                @(posedge clk);
                #1;

                timeout = timeout + 1;

                if (timeout > 30) begin
                    $fatal(1, "AXI RVALID timeout");
                end
            end
        end

        data = rdata;

        check_true(
            "AXI RRESP OKAY",
            rresp == 2'b00
        );

        @(negedge clk);
        arvalid = 1'b0;

        @(posedge clk);
        #1;
    end
endtask

task automatic wait_busy;
    reg [31:0] status;
    integer timeout;

    begin
        timeout = 0;
        status = 32'd0;

        while ((status & 32'h1) == 0) begin
            axi_read(8'h04, status);

            timeout = timeout + 1;

            if (timeout > 20) begin
                $fatal(1, "BUSY did not assert");
            end
        end
    end
endtask

task automatic wait_done;
    reg [31:0] status;
    integer timeout;

    begin
        timeout = 0;
        status = 32'd0;

        while ((status & 32'h2) == 0) begin
            axi_read(8'h04, status);

            timeout = timeout + 1;

            if (timeout > 50) begin
                $fatal(1, "DONE did not assert");
            end
        end
    end
endtask

task automatic wait_irq_high;
    integer timeout;

    begin
        timeout = 0;

        while (irq !== 1'b1) begin
            @(posedge clk);
            #1;

            timeout = timeout + 1;

            if (timeout > 80) begin
                $fatal(1, "IRQ did not assert");
            end
        end
    end
endtask

        "DONE W1C deasserts IRQ",
        irq === 1'b0

    //    enable ERROR only, then START while BUSY.

    axi_write(8'h00, 32'h0000_0006, 4'b0001, 1);

    #1;

        irq === 1'b1
    );

    check_true(
        "IRQ_STATUS ERROR pending",
        (rd & 32'h2) != 0

    axi_read(8'h04, rd);
    check_true(
        "STATUS ERROR sticky",
        (rd & 32'h4) != 0
